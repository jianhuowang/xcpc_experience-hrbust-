int n,e[maxn][maxn],d[maxn];
int Q[maxn];
void bfs(int s) {
    int l=1,r=0; //队首队尾指针
    for(int i=1;i<=n;i++) d[i]=-1; //d[i]=-1表示未访问过i
    d[s]=0;Q[++r]=s; //将d[s]设为0并将s加入队列中
    while(l<r) {
        int x=Q[l++]; //取出队首元素并删除。
        for(int i=1;i<=n;i++) if(e[x][i]) { //如果存在x到i的有向边
            if(d[i]<0) { //如果i未被访问过
                d[i]=d[x]+1; //将d[i]设为d[s]+1
                Q[++r]=i; //将i加入队列中
            }
        }
    }
}
